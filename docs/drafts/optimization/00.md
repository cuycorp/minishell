## ✅ Step 1: Define `t_exec_context`

Put this in a shared header (e.g., `executor.h`):

```c
typedef struct s_exec_context
{
	bool	is_in_pipeline;   // True if current command is part of a pipe
	bool	is_subshell;      // True if executing inside a subshell
	bool	should_fork;      // False if builtin must be run in parent
	int		stdin_fd;         // Input file descriptor
	int		stdout_fd;        // Output file descriptor (future use)
}	t_exec_context;
```

## ✅ Step 2: Update `ft_execute_ast_tree` to use `t_exec_context`

```c
int	ft_execute_ast_tree(t_ast_node *node, t_shell *data,
		t_exec_context *ctx, int *last_pid)
{
	int	result;

	if (!node || !data || !ctx)
		return (EXIT_FAILURE);

	if (node->type == AST_PIPE)
		return (ft_exec_pipe_node(node, data, ctx, last_pid));
	else if (node->type == AST_SIMPLE_COMMAND)
		return (ft_exec_simple_command(node->command_data, data, ctx, last_pid));
	else if (node->type == AST_LOGICAL_AND)
	{
		result = ft_execute_ast_tree(node->left, data, ctx, last_pid);
		if (result == EXIT_SUCCESS)
			return ft_execute_ast_tree(node->right, data, ctx, last_pid);
		return result;
	}
	else if (node->type == AST_LOGICAL_OR)
	{
		result = ft_execute_ast_tree(node->left, data, ctx, last_pid);
		if (result == EXIT_FAILURE)
			return ft_execute_ast_tree(node->right, data, ctx, last_pid);
		return result;
	}
	return EXIT_FAILURE;
}
```

## ✅ Step 3: Update `ft_exec_pipe_node` to create contexts for each side

```c
int	ft_exec_pipe_node(t_ast_node *root, t_shell *data,
		t_exec_context *ctx, int *last_pid)
{
	int		pipe_fd[2];
	pid_t	pid;
	int		result;

	if (!root || !data || !ctx)
		return (EXIT_FAILURE);

	if (pipe(pipe_fd) == -1)
		return (perror("pipe failed"), EXIT_FAILURE);

	pid = fork();
	if (pid < 0)
		return (perror("fork failed"), EXIT_FAILURE);

	if (pid == 0)
	{
		close(pipe_fd[0]); // Close read end

		if (dup2(pipe_fd[1], STDOUT_FILENO) == -1)
			exit(EXIT_FAILURE);
		close(pipe_fd[1]);

		if (ctx->stdin_fd != STDIN_FILENO)
		{
			dup2(ctx->stdin_fd, STDIN_FILENO);
			close(ctx->stdin_fd);
		}

		t_exec_context left_ctx = *ctx;
		left_ctx.is_in_pipeline = true;
		ft_execute_ast_tree(root->left, data, &left_ctx, last_pid);
		exit(EXIT_FAILURE); // fallback
	}

	// Parent
	close(pipe_fd[1]); // Close write end
	if (ctx->stdin_fd != STDIN_FILENO)
		close(ctx->stdin_fd);

	if (data->pid_count < data->command_count)
		data->pids[data->pid_count++] = pid;

	t_exec_context right_ctx = *ctx;
	right_ctx.is_in_pipeline = true;
	right_ctx.stdin_fd = pipe_fd[0];

	result = ft_execute_ast_tree(root->right, data, &right_ctx, last_pid);
	close(pipe_fd[0]);
	return result;
}
```

## ✅ Step 4: Update `ft_exec_simple_command` to use `t_exec_context`

```c
int	ft_exec_simple_command(t_command *command, t_shell *data,
		t_exec_context *ctx, int *last_pid)
{
	pid_t	pid;
	int		exit_code;

	if (!command || !data || !ctx || !last_pid)
		return (EXIT_FAILURE);

	bool is_parent_builtin = ft_is_parent_builtin(command->name);
	bool run_in_parent = is_parent_builtin && !ctx->is_in_pipeline;

	if (run_in_parent)
	{
		if (!ft_setup_redirection(command->redirection))
			return (EXIT_FAILURE);
		exit_code = ft_exec_parent_builtin(command, data);
		*last_pid = -1;
		return (exit_code);
	}

	pid = fork();
	if (pid < 0)
	{
		if (ctx->stdin_fd != STDIN_FILENO)
			close(ctx->stdin_fd);
		return (perror("fork failed"), EXIT_FAILURE);
	}

	if (pid == 0)
	{
		// CHILD
		if (ctx->stdin_fd != STDIN_FILENO)
		{
			if (dup2(ctx->stdin_fd, STDIN_FILENO) == -1)
				exit(EXIT_FAILURE);
			close(ctx->stdin_fd);
		}

		if (!ft_setup_redirection(command->redirection))
			exit(EXIT_FAILURE);

		if (ft_is_builtins(command->name))
			exit(ft_exec_builtin(command, data));
		else
			exit(ft_exec_command(command, data));
	}

	// PARENT
	if (ctx->stdin_fd != STDIN_FILENO)
		close(ctx->stdin_fd);

	if (data->pid_count < data->command_count)
	{
		data->pids[data->pid_count++] = pid;
		*last_pid = pid;
	}
	return (EXIT_SUCCESS);
}
```

## ✅ Step 5: Update `ft_executor`

```c
int	ft_executor(t_ast_node *root, t_shell *data, int stdin_fd, int *last_pid)
{
	t_exec_context ctx;

	ctx.stdin_fd = stdin_fd;
	ctx.stdout_fd = STDOUT_FILENO; // future extension
	ctx.is_in_pipeline = false;
	ctx.is_subshell = false;
	ctx.should_fork = true;

	if (!root || !data)
		return (EXIT_FAILURE);

	if (ft_execute_ast_tree(root, data, &ctx, last_pid) != 0)
		return (EXIT_FAILURE);
	return (EXIT_SUCCESS);
}
```
