## ✅ 3. `ft_execute_pipe_node`

Here’s the updated version with **full integration** of `stdout_fd` and `should_fork`:

```c
int	ft_execute_pipe_node(t_ast_node *node, t_shell *data,
		t_exec_context *ctx, int *last_pid)
{
	int		pipefd[2];
	pid_t	pid;
	t_exec_context left_ctx;
	t_exec_context right_ctx;

	if (!node || !node->left || !node->right)
		return (EXIT_FAILURE);
	if (pipe(pipefd) < 0)
		return (perror("pipe"), EXIT_FAILURE);

	// Prepare left side (write end of pipe)
	left_ctx = *ctx;
	left_ctx.stdout_fd = pipefd[1];
	left_ctx.is_in_pipeline = true;
	left_ctx.should_fork = true;

	ft_execute_ast_tree(node->left, data, &left_ctx, last_pid);
	close(pipefd[1]);

	// Prepare right side (read end of pipe)
	right_ctx = *ctx;
	right_ctx.stdin_fd = pipefd[0];
	right_ctx.is_in_pipeline = true;
	right_ctx.should_fork = true;

	ft_execute_ast_tree(node->right, data, &right_ctx, last_pid);
	close(pipefd[0]);

	return EXIT_SUCCESS;
}
```

✅ Clean and decoupled — no more duplicated pipe logic inside child or redirection logic.

## ✅ 5. `ft_handle_shell` — Top-level entry

Only sets up the initial `t_exec_context`.

int	ft_handle_shell(t_shell *data)
{
	t_exec_context	ctx;
	int				last_pid;

	ctx.stdin_fd = STDIN_FILENO;
	ctx.stdout_fd = STDOUT_FILENO;
	ctx.is_in_pipeline = false;
	ctx.is_subshell = false;
	ctx.should_fork = true;

	return ft_execute_ast_tree(data->ast_root, data, &ctx, &last_pid);
}

## ✅ BONUS: `ft_exec_subshell_node(...)`

```c
int	ft_execute_subshell_node(t_ast_node *node, t_shell *data,
		t_exec_context *ctx, int *last_pid)
{
	pid_t	pid;
	t_exec_context subctx;

	subctx = *ctx;
	subctx.is_subshell = true;
	subctx.should_fork = true;

	pid = fork();
	if (pid < 0)
		return (perror("fork"), EXIT_FAILURE);
	if (pid == 0)
	{
		if (subctx.stdin_fd != STDIN_FILENO)
			dup2(subctx.stdin_fd, STDIN_FILENO);
		if (subctx.stdout_fd != STDOUT_FILENO)
			dup2(subctx.stdout_fd, STDOUT_FILENO);
		ft_execute_ast_tree(node->left, data, &subctx, last_pid);
		exit(EXIT_SUCCESS);
	}

	if (ctx->stdin_fd != STDIN_FILENO)
		close(ctx->stdin_fd);
	if (ctx->stdout_fd != STDOUT_FILENO)
		close(ctx->stdout_fd);

	if (data->pid_count < data->command_count)
		data->pids[data->pid_count++] = pid;
	*last_pid = pid;
	return EXIT_SUCCESS;
}
```
