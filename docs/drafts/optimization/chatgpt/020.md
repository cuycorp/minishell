# Answer

You're absolutely on the right track thinking about introducing a centralized **execution context** (`t_exec_context`). It's one of the most **effective and scalable** ways to refactor execution logic — especially when dealing with complex pipelines, redirections, and conditional execution (`&&`, `||`, etc.).

---

## ✅ Is `t_exec_context` a Good Idea?

Yes — **definitely**.

### 🔥 Benefits of `t_exec_context`

* ✅ **Encapsulates execution state** in one place.
* ✅ **Simplifies function signatures** — you pass `t_exec_context *ctx` instead of many parameters.
* ✅ **Enables future extensions** (e.g. environment isolation, subshell flags, return codes).
* ✅ **Reduces side-effects and repeated logic** (like checking `stdin`, `stdout`, or needing to fork).

---

## 🤔 About `t_fd_backup`

You're right to be skeptical. The `t_fd_backup` idea is only useful if:

* You're doing **redirections in the parent shell process** (e.g., for builtins).
* You want to **restore original stdin/stdout after temporary dup2**.

If you're:

* Always forking for commands (except `cd`, `exit`, `export`, etc.)
* Applying redirections **only inside child**

Then ➤ you **don’t need `t_fd_backup`**. It just complicates the design.

---

## ✅ Final Structure

Let’s go with this:

```c
typedef struct s_exec_context {
	t_shell *shell;
	int input_fd;
	int output_fd;
	bool needs_fork;
	bool is_last_command;
} t_exec_context;
```

### 2. ✅ Refactor `ft_exec_simple_command`

```c
int ft_exec_simple_command(t_command *cmd, t_exec_context *ctx, int *last_pid)
{
	pid_t pid;

	if (!cmd || !ctx || !ctx->shell || !last_pid)
		return EXIT_FAILURE;

	if (ft_is_parent_builtins(cmd->name) && ctx->input_fd == STDIN_FILENO)
	{
		if (!ft_prepare_command_io(cmd->redirection))
			return EXIT_FAILURE;
		*last_pid = -1;
		return ft_exec_parent_builtin(cmd, ctx->shell);
	}

	// FORK child
	pid = fork();
	if (pid < 0)
		return (perror("fork"), EXIT_FAILURE);

	if (pid == 0) // child
	{
		if (ctx->input_fd != STDIN_FILENO)
			ft_apply_dup2(ctx->input_fd, STDIN_FILENO);
		if (ctx->output_fd != STDOUT_FILENO)
			ft_apply_dup2(ctx->output_fd, STDOUT_FILENO);
		if (!ft_prepare_command_io(cmd->redirection))
			exit(EXIT_FAILURE);
		if (ft_is_builtin_function(cmd->name))
			exit(ft_exec_builtin(cmd, ctx->shell));
		exit(ft_exec_command(cmd, ctx->shell));
	}

	if (ctx->input_fd != STDIN_FILENO)
		close(ctx->input_fd);
	if (ctx->output_fd != STDOUT_FILENO)
		close(ctx->output_fd);

	ctx->shell->pids[ctx->shell->pid_count++] = pid;
	*last_pid = pid;
	return EXIT_SUCCESS;
}
```

int ft_exec_pipe_node(t_ast_node *node, t_exec_context *ctx, int *last_pid)
{
	int pipe_fd[2];
	pid_t pid;

	if (pipe(pipe_fd) == -1)
		return (perror("pipe"), EXIT_FAILURE);

	pid = fork();
	if (pid < 0)
	{
		close(pipe_fd[0]);
		close(pipe_fd[1]);
		return (perror("fork"), EXIT_FAILURE);
	}

	if (pid == 0)
	{
		close(pipe_fd[0]);
		t_exec_context left_ctx = *ctx;
		left_ctx.output_fd = pipe_fd[1];
		exit(ft_execute_ast(node->left, &left_ctx, last_pid));
	}

	close(pipe_fd[1]);
	if (ctx->input_fd != STDIN_FILENO)
		close(ctx->input_fd);

	ctx->shell->pids[ctx->shell->pid_count++] = pid;

	t_exec_context right_ctx = *ctx;
	right_ctx.input_fd = pipe_fd[0];
	return ft_execute_ast(node->right, &right_ctx, last_pid);
}

### 4. ✅ Refactor `ft_execute_ast`

```c
int ft_execute_ast(t_ast_node *node, t_exec_context *ctx, int *last_pid)
{
	if (!node || !ctx)
		return EXIT_FAILURE;

	switch (node->type)
	{
		case AST_SIMPLE_COMMAND:
			return ft_exec_simple_command(node->command_data, ctx, last_pid);
		case AST_PIPE:
			return ft_exec_pipe_node(node, ctx, last_pid);
		case AST_LOGICAL_AND:
			if (ft_execute_ast(node->left, ctx, last_pid) == EXIT_SUCCESS)
				return ft_execute_ast(node->right, ctx, last_pid);
			return EXIT_FAILURE;
		case AST_LOGICAL_OR:
			if (ft_execute_ast(node->left, ctx, last_pid) != EXIT_SUCCESS)
				return ft_execute_ast(node->right, ctx, last_pid);
			return EXIT_SUCCESS;
		case AST_REDIRECTION:
			return ft_prepare_command_io(node->redirection_data)
				? EXIT_SUCCESS : EXIT_FAILURE;
		default:
			return EXIT_FAILURE;
	}
}
```

### 5. ✅ Update `ft_executor`

```c
int ft_executor(t_ast_node *root, t_shell *shell)
{
	t_exec_context ctx = {
		.shell = shell,
		.input_fd = STDIN_FILENO,
		.output_fd = STDOUT_FILENO,
		.needs_fork = true,
		.is_last_command = true,
	};
	int last_pid = -1;

	if (!ft_prepare_all_heredocs(root))
		return EXIT_FAILURE;

	shell->command_count = ft_count_cmd_nodes(root);
	shell->pids = malloc(sizeof(pid_t) * shell->command_count);
	if (!shell->pids)
		return (perror("malloc"), EXIT_FAILURE);

	return ft_execute_ast(root, &ctx, &last_pid);
}
```
