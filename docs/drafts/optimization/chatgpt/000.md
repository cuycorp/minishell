## 🔧 Refactoring Strategy Overview

### 🧠 **Main Goal**:

**Unify redirection logic into a clear, centralized and linear system** that’s easy to trace and scales well with logical operators, pipes, and simple commands.

## ✅ **Refactor Plan: Key Principles**

### 1. **Introduce a unified `ft_prepare_command_io`**

This function will:

* Process heredocs
* Open input/output files
* Perform `dup2`
* Handle errors

This replaces scattered:

* `ft_process_heredocs`
* `ft_setup_redirection`
* `ft_exec_redirections`
* `ft_handle_inputs`

---

### 2. **Simplify AST Execution**

Unify the logic in `ft_execute_ast_tree`:

* Handle IO *before* forking, using `ft_prepare_command_io`
* Reduce special cases (like redirection-only node)
* Let each node type (pipe, logic op, simple command) call a **common handler** for IO.

{
	int last_pid = -1;
	int result;

		return (EXIT_FAILURE);

	data->pids = malloc(sizeof(pid_t) * ft_count_cmd_nodes(root));
	if (!data->pids)
		return (perror("malloc failed"), EXIT_FAILURE);

	result = ft_execute_ast(root, data, STDIN_FILENO, &last_pid);
	return ft_wait_all_pids(data, &last_pid);
}

bool ft_prepare_all_heredocs(t_ast_node *node)

	{

## 🔧 Step 2: Centralize `ft_prepare_command_io`

```c
bool ft_prepare_command_io(t_redirection *redir)
{
	int in_fd = -1, out_fd = -1;
	t_redirection *r = redir;
	t_redirection *last_in = NULL;

	// Identify the last input redirection (needed for `<< a < b`)
	while (r)
	{
		if (r->type == HEREDOC || r->type == REDIRECT_IN)
			last_in = r;
		r = r->next;
	}
	r = redir;
	while (r)
	{
		if (r->type == HEREDOC)
		{
			if (r->heredoc_fd == -1)
				return false;
			if (r == last_in)
				in_fd = r->heredoc_fd;
			else
				close(r->heredoc_fd);
		}
		else if (r->type == REDIRECT_IN)
		{
			int fd = open(r->target, O_RDONLY);
			if (fd < 0)
				return (perror(r->target), false);
			if (r == last_in)
				in_fd = fd;
			else
				close(fd);
		}
		else if (r->type == REDIRECT_OUT)
		{
			out_fd = open(r->target, O_WRONLY | O_CREAT | O_TRUNC, 0644);
			if (out_fd < 0)
				return (perror(r->target), false);
		}
		else if (r->type == APPEND_OUT)
		{
			out_fd = open(r->target, O_WRONLY | O_CREAT | O_APPEND, 0644);
			if (out_fd < 0)
				return (perror(r->target), false);
		}
		r = r->next;
	}

	if (in_fd != -1 && dup2(in_fd, STDIN_FILENO) == -1)
		return (perror("dup2 in"), close(in_fd), false);
	if (out_fd != -1 && dup2(out_fd, STDOUT_FILENO) == -1)
		return (perror("dup2 out"), close(out_fd), false);
	if (in_fd != -1)
		close(in_fd);
	if (out_fd != -1)
		close(out_fd);
	return true;
}
```

## 🔧 Step 3: Clean `ft_exec_simple_command`

```c
int ft_exec_simple_command(t_command *cmd, t_shell *data, int input_fd, int *last_pid)
{
	pid_t pid;

	if (ft_is_parent_builtins(cmd->name) && input_fd == STDIN_FILENO)
	{
		if (!ft_prepare_command_io(cmd->redirection))
			return (EXIT_FAILURE);
		return ft_exec_parent_builtin(cmd, data);
	}

	pid = fork();
	if (pid < 0)
		return (perror("fork"), EXIT_FAILURE);
	if (pid == 0)
	{
		if (input_fd != STDIN_FILENO)
		{
			dup2(input_fd, STDIN_FILENO);
			close(input_fd);
		}
		if (!ft_prepare_command_io(cmd->redirection))
			exit(EXIT_FAILURE);
		if (ft_is_builtin_function(cmd->name))
			exit(ft_exec_builtin(cmd, data));
		exit(ft_exec_command(cmd, data));
	}
	if (input_fd != STDIN_FILENO)
		close(input_fd);
	data->pids[data->pid_count++] = pid;
	*last_pid = pid;
	return EXIT_SUCCESS;
}
```
