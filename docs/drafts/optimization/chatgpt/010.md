## 🔧 Refactored `ft_exec_pipe_node`

```c
int ft_exec_pipe_node(t_ast_node *node, t_shell *data, int input_fd, int *last_pid)
{
	int pipe_fd[2];
	pid_t pid;

	if (!node || pipe(pipe_fd) == -1)
		return (perror("pipe"), EXIT_FAILURE);

	pid = fork();
	if (pid < 0)
	{
		close(pipe_fd[0]);
		close(pipe_fd[1]);
		return (perror("fork"), EXIT_FAILURE);
	}

	if (pid == 0)
	{
		// Left side of pipe: stdout -> pipe, stdin -> input_fd
		close(pipe_fd[0]);
		if (!ft_apply_dup2(pipe_fd[1], STDOUT_FILENO))
			exit(EXIT_FAILURE);
		close(pipe_fd[1]);

		if (input_fd != STDIN_FILENO)
			ft_apply_dup2(input_fd, STDIN_FILENO);

		exit(ft_execute_ast(node->left, data, STDIN_FILENO, last_pid));
	}

	close(pipe_fd[1]); // Parent: keep read-end for next command
	if (input_fd != STDIN_FILENO)
		close(input_fd);

	// Track child PID
	if (data->pid_count < data->command_count)
		data->pids[data->pid_count++] = pid;

	// Right side: stdin <- pipe_fd[0]
	int result = ft_execute_ast(node->right, data, pipe_fd[0], last_pid);
	close(pipe_fd[0]);
	return result;
}
```

## 🔁 Updated `ft_execute_ast`

This becomes a **clean dispatcher**, no logic inside:

```c
int ft_execute_ast(t_ast_node *node, t_shell *data, int input_fd, int *last_pid)
{
	if (!node || !data)
		return EXIT_FAILURE;

	if (node->type == AST_PIPE)
		return ft_exec_pipe_node(node, data, input_fd, last_pid);
	else if (node->type == AST_SIMPLE_COMMAND)
		return ft_exec_simple_command(node->command_data, data, input_fd, last_pid);
	else if (node->type == AST_LOGICAL_AND)
		return ft_exec_logical_and(node, data, input_fd, last_pid);
	else if (node->type == AST_LOGICAL_OR)
		return ft_exec_logical_or(node, data, input_fd, last_pid);
	else if (node->type == AST_REDIRECTION)
	{
		if (!ft_prepare_command_io(node->redirection_data))
			return EXIT_FAILURE;
		return EXIT_SUCCESS; // Nothing to execute further
	}
	return EXIT_FAILURE;
}
```
