/* ============================
** File: executor.c
** Description: Updated executor code using t_exec_context
** ============================ */

#include "minishell.h"

/* Execution context struct */
typedef struct s_exec_context {
	t_shell		*shell;
	int			input_fd;
	int			output_fd;
	pid_t		*pids;
	int			pid_count;
	int			command_count;
	bool		needs_fork;
	bool		is_last_command;
} t_exec_context;

/* Process a heredoc input */
int ft_process_heredoc(const char *delimiter)
{
	int pipefd[2];
	char *line;

	if (pipe(pipefd) == -1)
		return (ft_perror("heredoc pipe failed"), -1);

	while (1)
	{
		line = readline("> ");
		if (!line || ft_strcmp(line, delimiter) == 0)
		{
			free(line);
			break;
		}
		write(pipefd[1], line, ft_strlen(line));
		write(pipefd[1], "\n", 1);
		free(line);
	}
	close(pipefd[1]);
	return pipefd[0];
}

/* Prepares command IO based on redirections */
void ft_prepare_command_io(t_redirection *redir, t_exec_context *ctx)
{
	t_redirection *curr;

	curr = redir;
	while (curr)
	{
		if (curr->type == REDIR_INPUT)
		{
			if (ctx->input_fd != STDIN_FILENO)
				close(ctx->input_fd);
			ctx->input_fd = open(curr->file, O_RDONLY);
		}
		else if (curr->type == REDIR_OUTPUT)
		{
			if (ctx->output_fd != STDOUT_FILENO)
				close(ctx->output_fd);
			ctx->output_fd = open(curr->file, O_CREAT | O_WRONLY | O_TRUNC, 0644);
		}
		else if (curr->type == REDIR_APPEND)
		{
			if (ctx->output_fd != STDOUT_FILENO)
				close(ctx->output_fd);
			ctx->output_fd = open(curr->file, O_CREAT | O_WRONLY | O_APPEND, 0644);
		}
		else if (curr->type == REDIR_HEREDOC)
		{
			if (ctx->input_fd != STDIN_FILENO)
				close(ctx->input_fd);
			ctx->input_fd = ft_process_heredoc(curr->delimiter);
		}
		curr = curr->next;
	}
}

/* Function to execute a simple command */
int ft_exec_simple_command(t_command *command, t_exec_context *ctx)
{
	pid_t pid;

	if (!command)
		return (EXIT_FAILURE);

	ft_prepare_command_io(command->redirection, ctx);

	if (ctx->needs_fork)
	{
		pid = fork();
		if (pid < 0)
			return (ft_perror("fork failed"), EXIT_FAILURE);
		if (pid == 0)
		{
			if (ctx->input_fd != STDIN_FILENO)
			{
				dup2(ctx->input_fd, STDIN_FILENO);
				close(ctx->input_fd);
			}
			if (ctx->output_fd != STDOUT_FILENO)
			{
				dup2(ctx->output_fd, STDOUT_FILENO);
				close(ctx->output_fd);
			}
			if (ft_is_builtin_function(command->name))
				exit(ft_exec_builtin(command, ctx->shell));
			else
				exit(ft_exec_command(command, ctx->shell));
		}
		ctx->pids[ctx->pid_count] = pid;
		ctx->pid_count = ctx->pid_count + 1;
	}
	else
	{
		if (ctx->input_fd != STDIN_FILENO)
		{
			dup2(ctx->input_fd, STDIN_FILENO);
			close(ctx->input_fd);
		}
		if (ctx->output_fd != STDOUT_FILENO)
		{
			dup2(ctx->output_fd, STDOUT_FILENO);
			close(ctx->output_fd);
		}
		return ft_exec_builtin(command, ctx->shell);
	}
	return EXIT_SUCCESS;
}

/* Function to execute a pipeline */
int ft_exec_pipe_node(t_ast_node *node, t_exec_context *ctx)
{
	int pipefd[2];
	t_exec_context left_ctx;
	t_exec_context right_ctx;

	if (pipe(pipefd) == -1)
		return (ft_perror("pipe failed"), EXIT_FAILURE);

	left_ctx = *ctx;
	right_ctx = *ctx;

	left_ctx.output_fd = pipefd[1];
	left_ctx.is_last_command = false;

	right_ctx.input_fd = pipefd[0];

	ft_execute_ast_tree(node->left, &left_ctx);
	close(pipefd[1]);

	ft_execute_ast_tree(node->right, &right_ctx);
	close(pipefd[0]);

	return EXIT_SUCCESS;
}

/* Entry point for executing AST */
int ft_execute_ast_tree(t_ast_node *node, t_exec_context *ctx)
{
	if (!node)
		return EXIT_FAILURE;

	if (node->type == AST_COMMAND)
		return ft_exec_simple_command(node->command_data, ctx);
	else if (node->type == AST_PIPE)
		return ft_exec_pipe_node(node, ctx);

	return EXIT_FAILURE;
}

/* Main executor entry point */
int ft_executor(t_ast_node *root, t_shell *shell)
{
	pid_t pids[256];
	t_exec_context ctx;
	int i;

	ctx.shell = shell;
	ctx.input_fd = STDIN_FILENO;
	ctx.output_fd = STDOUT_FILENO;
	ctx.pids = pids;
	ctx.pid_count = 0;
	ctx.command_count = ft_count_commands(root);
	ctx.needs_fork = true;
	ctx.is_last_command = true;

	ft_execute_ast_tree(root, &ctx);

	i = 0;
	while (i < ctx.pid_count)
	{
		waitpid(ctx.pids[i], NULL, 0);
		i = i + 1;
	}
	return 0;
}
