typedef struct s_fd_backup {
	int stdin_backup;
	int stdout_backup;
	int stderr_backup;
} t_fd_backup;

typedef struct s_exec_context {
	t_shell *shell;
	int input_fd;
	int output_fd;
	t_fd_backup fd_backup;
	bool needs_fork;
	bool is_last_command;
} t_exec_context;

// Backup original file descriptors
static bool ft_backup_fds(t_fd_backup *backup)
{
	backup->stdin_backup = dup(STDIN_FILENO);
	backup->stdout_backup = dup(STDOUT_FILENO);
	backup->stderr_backup = dup(STDERR_FILENO);

	return (backup->stdin_backup != -1 &&
			backup->stdout_backup != -1 &&
			backup->stderr_backup != -1);
}

// Restore original file descriptors
static void ft_restore_fds(t_fd_backup *backup)
{
	if (backup->stdin_backup != -1) {
		dup2(backup->stdin_backup, STDIN_FILENO);
		close(backup->stdin_backup);
	}
	if (backup->stdout_backup != -1) {
		dup2(backup->stdout_backup, STDOUT_FILENO);
		close(backup->stdout_backup);
	}
	if (backup->stderr_backup != -1) {
		dup2(backup->stderr_backup, STDERR_FILENO);
		close(backup->stderr_backup);
	}
}

// Process all heredocs in the AST before execution
static bool ft_process_all_heredocs(t_ast_node *node)
{
	if (!node)
		return (true);

	// Process current node's redirections
	if (node->type == AST_SIMPLE_COMMAND && node->command_data) {
		t_redirection *redir = node->command_data->redirection;
		while (redir) {
			if (redir->type == HEREDOC && redir->heredoc_fd == -1) {
				redir->heredoc_fd = ft_exec_heredoc(redir, redir->target);
				if (redir->heredoc_fd == -1)
					return (false);
			}
			redir = redir->next;
		}
	}

	// Recursively process children
	return (ft_process_all_heredocs(node->left) &&
			ft_process_all_heredocs(node->right));
}

// Unified redirection setup - handles all redirection types
static int ft_setup_redirections(t_redirection *redirections, int input_fd)
{
	t_redirection *last_input = NULL;
	int final_input_fd = input_fd;
	int fd;

	if (!redirections)
		return (input_fd);

	// Find the last input redirection (heredoc or redirect_in)
	t_redirection *current = redirections;
	while (current) {
		if (current->type == HEREDOC || current->type == REDIRECT_IN)
			last_input = current;
		current = current->next;
	}

	// Apply all redirections
	current = redirections;
	while (current) {
		switch (current->type) {
			case HEREDOC:
				if (current->heredoc_fd == -1)
					return (-1);
				if (current == last_input) {
					final_input_fd = current->heredoc_fd;
				} else {
					close(current->heredoc_fd);
					current->heredoc_fd = -1;
				}
				break;

			case REDIRECT_IN:
				fd = open(current->target, O_RDONLY);
				if (fd == -1) {
					ft_printf(STDERR_FILENO, "minishell: %s: %s\n",
							 current->target, strerror(errno));
					return (-1);
				}
				if (current == last_input) {
					final_input_fd = fd;
				} else {
					close(fd);
				}
				break;

			case REDIRECT_OUT:
				fd = open(current->target, O_WRONLY | O_CREAT | O_TRUNC, 0644);
				if (fd == -1) {
					ft_printf(STDERR_FILENO, "minishell: %s: %s\n",
							 current->target, strerror(errno));
					return (-1);
				}
				if (dup2(fd, STDOUT_FILENO) == -1) {
					close(fd);
					return (-1);
				}
				close(fd);
				break;

			case APPEND_OUT:
				fd = open(current->target, O_WRONLY | O_CREAT | O_APPEND, 0644);
				if (fd == -1) {
					ft_printf(STDERR_FILENO, "minishell: %s: %s\n",
							 current->target, strerror(errno));
					return (-1);
				}
				if (dup2(fd, STDOUT_FILENO) == -1) {
					close(fd);
					return (-1);
				}
				close(fd);
				break;
		}
		current = current->next;
	}

	// Apply input redirection if needed
	if (final_input_fd != STDIN_FILENO && final_input_fd != input_fd) {
		if (dup2(final_input_fd, STDIN_FILENO) == -1) {
			if (final_input_fd != input_fd)
				close(final_input_fd);
			return (-1);
		}
		if (final_input_fd != input_fd)
			close(final_input_fd);
	} else if (input_fd != STDIN_FILENO) {
		if (dup2(input_fd, STDIN_FILENO) == -1) {
			close(input_fd);
			return (-1);
		}
		close(input_fd);
	}

	return (0);
}

// Execute a simple command with proper context
static int ft_exec_simple_command_new(t_command *cmd, t_exec_context *ctx)
{
	pid_t pid;
	int exit_code;
	bool is_parent_builtin;

	if (!cmd || !ctx)
		return (EXIT_FAILURE);

	is_parent_builtin = ft_is_parent_builtins(cmd->name);

	// Handle parent builtins without fork if no pipe input
	if (is_parent_builtin && ctx->input_fd == STDIN_FILENO && !ctx->needs_fork) {
		t_fd_backup backup;
		if (!ft_backup_fds(&backup))
			return (EXIT_FAILURE);

		if (ft_setup_redirections(cmd->redirection, ctx->input_fd) == -1) {
			ft_restore_fds(&backup);
			return (EXIT_FAILURE);
		}

		exit_code = ft_exec_parent_builtin(cmd, ctx->shell);
		ft_restore_fds(&backup);
		return (exit_code);
	}

	// Fork for all other cases
	pid = fork();
	if (pid < 0) {
		if (ctx->input_fd != STDIN_FILENO)
			close(ctx->input_fd);
		return (EXIT_FAILURE);
	}

	if (pid == 0) {
		// Child process
		if (ft_setup_redirections(cmd->redirection, ctx->input_fd) == -1)
			exit(EXIT_FAILURE);

		if (ft_is_builtin_function(cmd->name))
			exit_code = ft_exec_builtin(cmd, ctx->shell);
		else
			exit_code = ft_exec_command(cmd, ctx->shell);

		exit(exit_code);
	}

	// Parent process
	if (ctx->input_fd != STDIN_FILENO)
		close(ctx->input_fd);

	// Store PID for waiting
	if (ctx->shell->pid_count < ctx->shell->command_count) {
		ctx->shell->pids[ctx->shell->pid_count] = pid;
		ctx->shell->pid_count++;
	}

	return (EXIT_SUCCESS);
}

// Execute pipe node with cleaner logic
static int ft_exec_pipe_new(t_ast_node *node, t_exec_context *ctx)
{
	int pipe_fd[2];
	pid_t pid;
	int result;
	t_exec_context left_ctx, right_ctx;

	if (!node || !ctx)
		return (EXIT_FAILURE);

	if (pipe(pipe_fd) == -1) {
		perror("pipe failed");
		return (EXIT_FAILURE);
	}

	// Setup context for left child (producer)
	left_ctx = *ctx;
	left_ctx.output_fd = pipe_fd[WRITE_END];
	left_ctx.needs_fork = true;
	left_ctx.is_last_command = false;

	pid = fork();
	if (pid < 0) {
		close(pipe_fd[READ_END]);
		close(pipe_fd[WRITE_END]);
		return (EXIT_FAILURE);
	}

	if (pid == 0) {
		// Left child process
		close(pipe_fd[READ_END]);
		if (dup2(pipe_fd[WRITE_END], STDOUT_FILENO) == -1)
			exit(EXIT_FAILURE);
		close(pipe_fd[WRITE_END]);

		if (ctx->input_fd != STDIN_FILENO) {
			if (dup2(ctx->input_fd, STDIN_FILENO) == -1)
				exit(EXIT_FAILURE);
			close(ctx->input_fd);
		}

		int exit_code = ft_execute_node(node->left, &left_ctx);
		exit(exit_code);
	}

	// Parent continues with right child
	close(pipe_fd[WRITE_END]);
	if (ctx->input_fd != STDIN_FILENO)
		close(ctx->input_fd);

	// Store left child PID
	if (ctx->shell->pid_count < ctx->shell->command_count) {
		ctx->shell->pids[ctx->shell->pid_count] = pid;
		ctx->shell->pid_count++;
	}

	// Setup context for right child (consumer)
	right_ctx = *ctx;
	right_ctx.input_fd = pipe_fd[READ_END];

	result = ft_execute_node(node->right, &right_ctx);
	close(pipe_fd[READ_END]);

	return (result);
}

// Main execution dispatcher
static int ft_execute_node(t_ast_node *node, t_exec_context *ctx)
{
	int result;

	if (!node || !ctx)
		return (EXIT_FAILURE);

	switch (node->type) {
		case AST_SIMPLE_COMMAND:
			return (ft_exec_simple_command_new(node->command_data, ctx));

		case AST_PIPE:
			return (ft_exec_pipe_new(node, ctx));

		case AST_LOGICAL_AND:
			result = ft_execute_node(node->left, ctx);
			if (result == EXIT_SUCCESS)
				return (ft_execute_node(node->right, ctx));
			return (result);

		case AST_LOGICAL_OR:
			result = ft_execute_node(node->left, ctx);
			if (result != EXIT_SUCCESS)
				return (ft_execute_node(node->right, ctx));
			return (result);

		case AST_REDIRECTION:
			// Handle standalone redirections
			return (ft_exec_standalone_redirections(node->redirection_data, ctx));

		default:
			return (EXIT_FAILURE);
	}
}

// Handle standalone redirections (no command)
static int ft_exec_standalone_redirections(t_redirection *redir, t_exec_context *ctx)
{
	pid_t pid;
	int status, exit_code;

	if (!redir)
		return (EXIT_FAILURE);

	pid = fork();
	if (pid < 0)
		return (EXIT_FAILURE);

	if (pid == 0) {
		if (ft_setup_redirections(redir, STDIN_FILENO) == -1)
			exit(EXIT_FAILURE);
		exit(EXIT_SUCCESS);
	}

	if (waitpid(pid, &status, 0) == -1)
		return (EXIT_FAILURE);

	if (WIFEXITED(status))
		exit_code = WEXITSTATUS(status);
	else if (WIFSIGNALED(status))
		exit_code = 128 + WTERMSIG(status);
	else
		exit_code = EXIT_FAILURE;

	return (exit_code);
}

// MAIN EXECUTION INTERFACE
// =============================================================================

int ft_execute_ast(t_ast_node *root, t_shell *shell)
{
	t_exec_context ctx;

	if (!root || !shell)
		return (EXIT_FAILURE);

	// Initialize execution context
	ctx.shell = shell;
	ctx.input_fd = STDIN_FILENO;
	ctx.output_fd = STDOUT_FILENO;
	ctx.needs_fork = false;
	ctx.is_last_command = true;

	// Process all heredocs first
	if (!ft_process_all_heredocs(root)) {
		ft_printf(STDERR_FILENO, "heredoc processing failed\n");
		return (EXIT_FAILURE);
	}

	// Count commands and allocate PID array
	shell->command_count = ft_count_cmd_nodes(root);
	if (shell->command_count > 0) {
		shell->pids = malloc(sizeof(pid_t) * shell->command_count);
		if (!shell->pids) {
			perror("malloc failed");
			return (EXIT_FAILURE);
		}
		shell->pid_count = 0;
	}

	// Execute the AST
	return (ft_execute_node(root, &ctx));
}

static void ft_close_all_heredocs(t_ast_node *node)
{
	if (!node)
		return;

	if (node->type == AST_SIMPLE_COMMAND && node->command_data) {
		t_redirection *redir = node->command_data->redirection;
		while (redir) {
			if (redir->type == HEREDOC && redir->heredoc_fd != -1) {
				close(redir->heredoc_fd);
				redir->heredoc_fd = -1;
			}
			redir = redir->next;
		}
	}

	ft_close_all_heredocs(node->left);
	ft_close_all_heredocs(node->right);
}

static void ft_reset_shell_new(t_shell *data)
{
	if (!data)
		return;

	if (data->input) {
		free(data->input);
		data->input = NULL;
	}

	if (data->tokens_list)
		ft_free_tokens_list(&data->tokens_list);

	if (data->ast_root) {
		ft_close_all_heredocs(data->ast_root);
		ft_free_ast_tree(&data->ast_root);
	}

	if (data->pids) {
		free(data->pids);
		data->pids = NULL;
	}

	data->pid_count = 0;
	data->command_count = 0;
}

// =============================================================================
// UPDATED MAIN SHELL LOOP
// =============================================================================

void ft_handle_shell_new(t_shell *data)
{
	int exit_code = 0;
	pid_t last_pid = -1;

	while (1) {
		// ... (parsing code remains the same)

		if (data->ast_root) {
			ft_print_ast_node(data->ast_root);

			// Execute the AST
			if (ft_execute_ast(data->ast_root, data) == EXIT_SUCCESS) {
				if (data->command_count > 0) {
					exit_code = ft_wait_all_pids(data, &last_pid);
				} else {
					exit_code = EXIT_SUCCESS;
				}
			} else {
				exit_code = EXIT_FAILURE;
				ft_printf(STDERR_FILENO, "Execution failed\n");
			}

			ft_printf(STDOUT_FILENO, "exit code = %d\n", exit_code);
		}

		ft_reset_shell_new(data);
	}
}
