2. Helper: replace current token with match list (returns the last inserted token on success, NULL on error)

```C
/* Replace the token `current` value with matches[0] and insert
   additional tokens for matches[1..]. Returns pointer to the last
   token inserted (which may be `current` if only one match),
   or NULL on error. */
static t_token	*ft_replace_current_token_with_matches(t_token *current, char **matches)
{
	t_token	*last;
	t_token	*first_new;
	t_token	*node;
	int		i;
	char	*tmp;
	t_token	*orig_next;

	if (!current || !matches || !matches[0])
		return (NULL);

	/* Duplicate first match and replace current->value */
	tmp = ft_strdup(matches[0]);
	if (!tmp)
		return (NULL);
	free(current->value);
	current->value = tmp;
	current->type = TOKEN_WORD; /* wildcard results are plain words */

	orig_next = current->next;
	last = current;
	first_new = NULL;

	/* Insert additional matches as new tokens (if any) */
	i = 1;
	while (matches[i])
	{
		node = malloc(sizeof(t_token));
		if (!node)
			break;
		node->value = ft_strdup(matches[i]);
		if (!node->value)
		{
			free(node);
			break;
		}
		node->type = TOKEN_WORD;
		node->next = last->next;
		last->next = node;
		if (!first_new)
			first_new = node;
		last = node;
		i++;
	}
	/* If we broke early due to allocation failure, roll back created nodes */
	if (matches[i])
	{
		/* free all newly created nodes starting from first_new (if any) */
		node = first_new;
		while (node)
		{
			t_token *next = node->next;
			free(node->value);
			free(node);
			if (node == last)
				break;
			node = next;
		}
		/* restore original next pointer */
		current->next = orig_next;
		return (NULL);
	}
	return (last);
}
```
