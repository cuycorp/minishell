/* Expand wildcards (glob) for unquoted tokens (TOKEN_WORD).
   Skips the heredoc delimiter (i.e. token immediately after TOKEN_HEREDOC). */
bool	ft_expand_wildcards_in_token_list(t_token **token_list)
{
	t_token	*current;
	char	**matches;
	t_token	*last;

	if (!token_list)
		return (false);
	current = *token_list;
	while (current)
	{
		/* If we see a heredoc token, skip its delimiter token (no wildcard
		   expansion for heredoc delimiters). Move current past the delimiter. */
		if (current->type == TOKEN_HEREDOC)
		{
			if (current->next)
				current = current->next->next;
			else
				current = NULL;
			continue;
		}

		/* Only expand unquoted words (plain tokens). */
		if (current->type == TOKEN_WORD && current->value && ft_has_wildcard(current->value))
		{
			matches = ft_resolve_wildcard_pattern(current->value);
			if (!matches)
				return (false); /* resolve failed (allocation or opendir error) */

			/* Replace current token with matches (1..n). ft_resolve_wildcard_pattern
			   returns an array containing at least one element (pattern if no match). */
			last = ft_replace_current_token_with_matches(current, matches);
			ft_free_char_tab(matches);
			if (!last)
				return (false); /* allocation failure during insertion */

			/* Continue after the last inserted token */
			current = last->next;
			continue;
		}
		current = current->next;
	}
	return (true);
}
