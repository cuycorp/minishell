#include "minishell.h"

// Replace one token with multiple matches in the token list
static void ft_replace_current_token_with_matches(t_token **tokens,
												  t_token **cur,
												  t_list *matches)
{
	t_token *prev;
	t_token *after;
	t_token *first_new;
	t_token *last_new;
	t_list  *node;

	prev = ft_get_prev_token(*tokens, *cur);
	after = (*cur)->next;

	// Build new tokens from matches
	node = matches;
	first_new = NULL;
	last_new = NULL;
	while (node)
	{
		t_token *new_tok = ft_new_token(TOKEN_WORD, ft_strdup(node->content), 0);
		if (!first_new)
			first_new = new_tok;
		else
			last_new->next = new_tok;
		last_new = new_tok;
		node = node->next;
	}

	// Link into list
	if (prev)
		prev->next = first_new;
	else
		*tokens = first_new;
	last_new->next = after;

	// Move cur to last inserted so loop continues correctly
	*cur = last_new;
}

// Handle wildcard expansion for a redirection target
static int ft_expand_wildcards_for_redirect(t_token *file_token)
{
	t_list *matches;

	matches = ft_resolve_wildcard_pattern(file_token->value);
	if (!matches)
		return (0); // No matches → keep as-is

	if (ft_lstsize(matches) > 1)
	{
		ft_putstr_fd("bash: ", 2);
		ft_putstr_fd(file_token->value, 2);
		ft_putendl_fd(": ambiguous redirect", 2);
		ft_lstclear(&matches, free);
		return (-1);
	}
	// One match → replace token value
	free(file_token->value);
	file_token->value = ft_strdup(matches->content);
	ft_lstclear(&matches, free);
	return (0);
}

// Main expansion
int ft_expand_wildcards_in_token_list(t_token **tokens)
{
	t_token *cur;
	int     ret;
	t_list  *matches;

	cur = *tokens;
	while (cur)
	{
		if (cur->type == TOKEN_WORD && !cur->quoted)
		{
			matches = ft_resolve_wildcard_pattern(cur->value);
			if (matches)
			{
				if (ft_lstsize(matches) > 1)
					ft_replace_current_token_with_matches(tokens, &cur, matches);
				else
				{
					// Single match → just replace value
					free(cur->value);
					cur->value = ft_strdup(matches->content);
				}
				ft_lstclear(&matches, free);
			}
		}
		else if (cur->type == TOKEN_REDIRECT_IN
			  || cur->type == TOKEN_REDIRECT_OUT
			  || cur->type == TOKEN_APPEND
			  || cur->type == TOKEN_HEREDOC)
		{
			if (cur->next && cur->next->type == TOKEN_WORD && !cur->next->quoted)
			{
				ret = ft_expand_wildcards_for_redirect(cur->next);
				if (ret < 0)
					return (-1);
			}
		}
		cur = cur->next;
	}
	return (0);
}
