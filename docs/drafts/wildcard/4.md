static int expand_wildcards_for_redirect(t_token *file_token)
{
	t_list *matches;

	matches = ft_match_wildcard(file_token->value);
	if (!matches)
		return (0); // No matches → keep token (bash will later fail if file doesn't exist)

	if (ft_lstsize(matches) > 1)
	{
		ft_putstr_fd("bash: ", 2);
		ft_putstr_fd(file_token->value, 2);
		ft_putendl_fd(": ambiguous redirect", 2);
		ft_lstclear(&matches, free);
		return (-1); // Error → abort
	}
	// Exactly one match → replace token value
	free(file_token->value);
	file_token->value = ft_strdup(matches->content);
	ft_lstclear(&matches, free);
	return (0);
}

int ft_expand_wildcards_in_token_list(t_token **tokens)
{
	t_token *cur = *tokens;
	int ret;

	while (cur)
	{
		if (cur->type == TOKEN_WORD && !cur->quoted)
		{
			// Expand normal command arguments
			ft_expand_wildcard_in_argument(cur);
		}
		else if (cur->type == TOKEN_REDIRECT_IN || cur->type == TOKEN_REDIRECT_OUT
				 || cur->type == TOKEN_APPEND || cur->type == TOKEN_HEREDOC)
		{
			if (cur->next && cur->next->type == TOKEN_WORD && !cur->next->quoted)
			{
				ret = expand_wildcards_for_redirect(cur->next);
				if (ret < 0)
					return (-1); // Stop everything
			}
		}
		cur = cur->next;
	}
	return (0);
}
