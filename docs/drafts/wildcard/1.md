1) Tiny change to ft_expansion_n_removal — call wildcard expansion after retokenize

```C
bool	ft_expansion_n_removal(t_shell *data)
{
	if (!data)
		return (false);
	if (!ft_expand_variables_in_token_list(&data->tokens_list, data))
		return (false);
	if (!ft_retokenize(data))
		return (false);

	/* NEW: expand wildcards (pathname expansion) after retokenize,
	   i.e. after field-splitting and before quote removal. */
	if (!ft_expand_wildcards_in_token_list(&data->tokens_list))
		return (false);

	if (!ft_quote_removal(&data->tokens_list))
		return (false);
	return (true);
}
```
